cmake_minimum_required(VERSION 3.16)
project(KeyValueStore VERSION 1.0.0 LANGUAGES CXX)
if(NOT UNIX)
    message(FATAL_ERROR "KeyValueStore requires POSIX APIs (Linux or macOS).")
endif()
find_package(Threads REQUIRED)
add_library(kv_store src/store.cpp)
target_include_directories(kv_store PUBLIC include)
target_compile_features(kv_store PUBLIC cxx_std_17)
set_target_properties(kv_store PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(kv_store PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(kv_store PUBLIC Threads::Threads)
add_executable(kv_cli src/cli.cpp)
target_link_libraries(kv_cli PRIVATE kv_store)
add_executable(kv_bench bench/benchmark.cpp)
target_link_libraries(kv_bench PRIVATE kv_store)
include(CTest)
if(BUILD_TESTING)
    add_executable(kv_tests tests/store_test.cpp)
    target_link_libraries(kv_tests PRIVATE kv_store)
    add_test(NAME store COMMAND kv_tests)
endif()
